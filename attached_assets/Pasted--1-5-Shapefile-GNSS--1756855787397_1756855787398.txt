تم. سأبدأ بإعداد نفس الواجهات (الصفحات 1–5) مع ربطها كاملاً بالمسارين (Shapefile/GNSS) وبما يتوافق مع نموذج “الفرع التنفيذي/المكتب الإشرافي”. أدناه مخطط التنقل، سقالات الشاشات، حالة المَعالج (Wizard) المشتركة، وربط REST APIs. يمكنك نسخ الكود كما هو وتوصيله.

1) خريطة التنقل والربط بين الشاشات
- /admin/login → شاشة دخول النظام الإداري (صفحة 1)
- /services → دليل الخدمات (صفحة 2)
- /survey/wizard
  - ?step=choose-path → اختيار المسار (جديد)
  - ?step=applicant → بيانات مقدم الطلب
  - ?step=location → تحديد الموقع الجغرافي (صفحة 3)
  - ?step=attachments → المرفقات (Shapefile أو مستندات GNSS) (صفحة 3)
  - ?step=billing → طباعة الفاتورة والسداد (صفحة 4)
  - بعد الدفع:
    - /cashier/awaiting → شاشة مسؤول الصندوق “طلبات بانتظار السداد”
    - /cashier/invoice/:invoiceId → تأكيد السداد (صفحة 5)

منطق اختيار المسار
- النظام يحاول كشف قرار سابق تلقائياً (Legacy lookup + Point-in-Polygon)
- إذا وُجد قرار سابق مقترن بالقطعة → المسار Shapefile
- إن لم يوجد → GNSS
- يمكن السماح بالاختيار اليدوي وفق سياسة الفرع (مع تحذير/تقييد)

import { createContext, useContext, useMemo, useState } from "react";

export type WorkflowMode = 'shapefile' | 'gnss';

export interface ApplicantData {
  nationalId?: string;
  fullName?: string;
  phone?: string;
  role?: 'self' | 'agent';
  documentType?: 'deed' | 'basira' | 'other';
  purpose?: string;
  areaFromDoc?: number;
}

export interface LocationData {
  governorate?: string;
  district?: string;
  subDistrict?: string;
  sector?: string;
  neighborhood?: string;
  planNo?: string;
  lng?: number;
  lat?: number;
  pnpContext?: any; // من خدمة PnP
  legacyFound?: boolean;
}

export interface WizardState {
  requestId?: string;
  step: 'choose-path'|'applicant'|'location'|'attachments'|'billing';
  mode?: WorkflowMode;
  applicant: ApplicantData;
  location: LocationData;
  attachments: {
    shapefileZip?: File | null;
    otherDocs?: File[];
    qc?: {
      crsOk?: boolean;
      topologyOk?: boolean;
      insidePlan?: boolean;
      areaDiffPct?: number;
      streetsMatched?: boolean;
      report?: any;
    }
  };
  invoiceId?: string;
}

const SurveyWizardCtx = createContext<{
  state: WizardState;
  setState: React.Dispatch<React.SetStateAction<WizardState>>;
} | null>(null);

export function SurveyWizardProvider({ children }: { children: React.ReactNode }) {
  const [state, setState] = useState<WizardState>({
    step: 'choose-path',
    applicant: {},
    location: {},
    attachments: {}
  });
  const value = useMemo(()=>({state, setState}),[state]);
  return <SurveyWizardCtx.Provider value={value}>{children}</SurveyWizardCtx.Provider>;
}
export function useSurveyWizard() {
  const ctx = useContext(SurveyWizardCtx);
  if (!ctx) throw new Error("useSurveyWizard must be used within provider");
  return ctx;
}

7) خطوة 2: تحديد الموقع الجغرافي (مع PnP وتبديل المسار إن لزم)
client/src/pages/survey/steps/LocationStep.tsx
import { useSurveyWizard } from "@/features/survey/wizard/SurveyWizardContext";
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { apiRequest } from "@/lib/queryClient";

export default function LocationStep() {
  const { state, setState } = useSurveyWizard();
  const [coords, setCoords] = useState({ lat: state.location.lat, lng: state.location.lng });

  const runPnP = async () => {
    const pnp = await apiRequest("POST", "/api/gis/point-in-polygon", { latitude: coords?.lat, longitude: coords?.lng });
    const legacyFound = await apiRequest("GET", `/api/survey/legacy/check?lng=${coords.lng}&lat=${coords.lat}`);
    setState(s => ({
      ...s,
      location: { ...s.location, lat: coords.lat, lng: coords.lng, pnpContext: pnp, legacyFound: !!legacyFound?.found },
      mode: legacyFound?.found ? 'shapefile' : s.mode
    }));
  };

  const next = async () => {
    await runPnP();
    setState(s => ({ ...s, step: 'attachments' }));
  };

  return (
    <div className="space-y-4">
      {/* ضع هنا مكوّن الخريطة الذي لديك */}
      <div className="grid md:grid-cols-2 gap-4">
        <input className="input" placeholder="المحافظة" value={state.location.governorate ?? ''} onChange={e=>setState(s=>({...s, location:{...s.location, governorate:e.target.value}}))}/>
        <input className="input" placeholder="المديرية" value={state.location.district ?? ''} onChange={e=>setState(s=>({...s, location:{...s.location, district:e.target.value}}))}/>
        <input className="input" placeholder="العزلة" onChange={e=>setState(s=>({...s, location:{...s.location, subDistrict:e.target.value}}))}/>
        <input className="input" placeholder="القطاع" onChange={e=>setState(s=>({...s, location:{...s.location, sector:e.target.value}}))}/>
        <input className="input" placeholder="رقم المخطط" onChange={e=>setState(s=>({...s, location:{...s.location, planNo:e.target.value}}))}/>
      </div>
      <div className="flex justify-between">
        <Button variant="outline" onClick={()=>setState(s=>({...s, step:'applicant'}))}>رجوع</Button>
        <Button onClick={next}>التالي</Button>
      </div>
    </div>
  );
}

8) خطوة 3: المرفقات (يتبدّل حسب المسار)
client/src/pages/survey/steps/AttachmentsStep.tsx
import { useSurveyWizard } from "@/features/survey/wizard/SurveyWizardContext";
import { useMutation } from "@tanstack/react-query";
import { apiRequest } from "@/lib/queryClient";
import { Button } from "@/components/ui/button";
import { useState } from "react";

export default function AttachmentsStep() {
  const { state, setState } = useSurveyWizard();
  const [file, setFile] = useState<File|null>(null);

  const uploadShp = useMutation({
    mutationFn: async (file: File) => {
      const form = new FormData();
      form.append("file", file);
      return fetch(`/api/survey/requests/${state.requestId ?? 'new'}/shp/upload`, { method: "POST", body: form })
        .then(r => r.json());
    },
    onSuccess: (data) => {
      setState(s => ({ ...s, attachments: { ...s.attachments, shapefileZip: file, qc: data?.qc }, step: 'billing' }));
    }
  });

  const uploadDocs = useMutation({
    mutationFn: async () => apiRequest("POST", `/api/survey/requests/${state.requestId ?? 'new'}/docs`, { /* وثائق الملكية... */ }),
    onSuccess: () => setState(s => ({ ...s, step: 'billing' }))
  });

  if (state.mode === 'shapefile') {
    return (
      <div className="space-y-4">
        <p className="text-sm text-gray-600">ارفع ملف ZIP يحتوي على (shp, shx, dbf, prj). سيقوم النظام بفحص CRS والطوبولوجيا والمساحة...</p>
        <input type="file" onChange={e => setFile(e.target.files?.[0] ?? null)} />
        <div className="flex justify-between">
          <Button variant="outline" onClick={()=>setState(s=>({...s, step:'location'}))}>رجوع</Button>
          <Button disabled={!file || uploadShp.isLoading} onClick={()=>file && uploadShp.mutate(file)}>رفع ومتابعة</Button>
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <p className="text-sm text-gray-600">أرفق وثيقة الملكية وأي مستندات إضافية. سيتم تحديد موعد الرفع الميداني.</p>
      {/* حقول رفع وثائق متعددة */}
      <div className="flex justify-between">
        <Button variant="outline" onClick={()=>setState(s=>({...s, step:'location'}))}>رجوع</Button>
        <Button onClick={()=>uploadDocs.mutate()} disabled={uploadDocs.isLoading}>حفظ ومتابعة</Button>
      </div>
    </div>
  );
}

11) ربط REST APIs (خلاصة)
- الكشف عن قرار سابق:
  GET /api/survey/legacy/check?planNo=..&lng=..&lat=..
  → { found: boolean, referenceId?:string }
- PnP:
  POST /api/gis/point-in-polygon {latitude, longitude}
- إنشاء/تحديث طلب:
  POST /api/survey/requests (يرد requestId)
  PUT  /api/survey/requests/:id
- Shapefile:
  POST /api/survey/requests/:id/shp/upload (form-data)
  GET  /api/survey/requests/:id/shp/validation
  POST /api/survey/requests/:id/shp/commit
- GNSS (قائم):
  POST /api/survey-requests/:id/points
  POST /api/survey-requests/:id/features
- الفوترة:
  POST /api/billing/invoices { requestId, mode }
  GET  /api/billing/invoices/:invoiceId
  POST /api/billing/invoices/:invoiceId/confirm { receiptNo, notes }
