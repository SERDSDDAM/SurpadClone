1) خدمة توليد PDF (decisionPdf.ts)
import puppeteer from 'puppeteer';
import QRCode from 'qrcode';
import Handlebars from 'handlebars';
import dayjs from 'dayjs';
import fs from 'fs/promises';
import path from 'path';
import { db } from '../db';
import { surveyRequests } from '../../shared/survey-schema';
import { eq } from 'drizzle-orm';

export async function generateDecisionPdf(requestId: string) {
  // 1) جلب البيانات الأساسية
  const [r] = await db.select().from(surveyRequests).where(eq(surveyRequests.id, requestId));
  if (!r) throw new Error('Survey request not found');

  // 2) إعداد بيانات العرض
  const decisionNumber = `QM-${new Date().getFullYear()}-${Date.now().toString().slice(-6)}`;
  const verifyUrl = `${process.env.PUBLIC_URL || 'https://example.gov.ye'}/verify/${decisionNumber}`;
  const qrDataUrl = await QRCode.toDataURL(verifyUrl, { width: 256 });

  // setbacks مبدئية حسب قرار المكتب (إن وجد)
  let setbacks = null as null | { front: number; sides: number; rear: number };
  const widthM = r.officeDecision?.widthM ?? null;
  if (widthM) setbacks = { front: widthM >= 12 ? 5 : widthM >= 8 ? 3 : 2, sides: 2, rear: 2 };

  // صور ثلاثية (خرائط) – اجعلها اختيارية أو من خدمة الصور لديك
  const mapImages = {
    plan: r.planImageUrl || null,
    aerial: r.aerialImageUrl || null,
    geometry: r.geometryPreviewUrl || null
  };

  // 3) توليد HTML من القالب
  const templatePath = path.join(process.cwd(), 'server', 'templates', 'decision.hbs');
  const template = await fs.readFile(templatePath, 'utf8');
  const compile = Handlebars.compile(template);

  const html = compile({
    header: {
      decisionNumber,
      date: dayjs().format('YYYY-MM-DD'),
      governorate: r.governorate,
      directorate: r.directorate,
      qrDataUrl
    },
    owner: { name: r.ownerName, phone: r.ownerPhone || '-' },
    request: {
      requestNumber: r.requestNumber,
      purpose: r.purpose,
      geometrySource: r.geometrySource,
      pnp: r.pnpContext
    },
    officeDecision: r.officeDecision || null,
    setbacks,
    // حدود/أبعاد إن كانت مخزنة لديك في boundaries
    boundaries: r.boundaries || [], // [{side:'N', length_m:..., type:'شارع', streetCode:'..', width:..}, ...]
    areas: {
      byPlan: r.areaPlan ?? r.area ?? null,
      byNature: r.areaNature ?? null
    },
    images: mapImages,
    footer: {
      engineer: r.engineerName || '—',
      office: 'مكتب الأشغال العامة والطرق',
      note: r.geometrySource === 'uploaded_shp' ? 'اعتماد عبر ملف شِيب فايل' : 'اعتماد عبر GNSS'
    }
  });

  // 4) تحويل HTML → PDF (RTL)
  const browser = await puppeteer.launch({ headless: 'new' });
  const page = await browser.newPage();
  await page.setContent(html, { waitUntil: 'networkidle0' });
  const pdf = await page.pdf({
    format: 'A4',
    printBackground: true,
    margin: { top: '10mm', right: '10mm', bottom: '12mm', left: '10mm' }
  });
  await browser.close();

  // 5) إعادة الـ Buffer + decisionNumber
  return { pdf, decisionNumber };
}

3) مسار التنزيل (Route)
router.get('/survey/decisions/:decisionNumber/download', async (req, res) => {
  try {
    // مبدئياً اقبل requestId عبر query أو استرجعه برابط داخلي
    const requestId = String(req.query.requestId || '');
    const { pdf } = await generateDecisionPdf(requestId);
    res.setHeader('Content-Type', 'application/pdf');
    res.setHeader('Content-Disposition', `inline; filename="decision-${req.params.decisionNumber}.pdf"`);
    res.send(pdf);
  } catch (e) {
    console.error('PDF download error:', e);
    res.status(500).json({ error: 'failed to download pdf' });
  }
});

خامساً: UAT رسمي (3 أيام) – ما سنفحصه
- اليوم 1:
  - مسار Shapefile: رفع → QC → commit → PDF.
  - مسار GNSS: نقاط/اكتمال → PDF.
- اليوم 2:
  - حالات التصعيد: street_status → قرار المكتب → PDF مع setbacks.
  - حالات حساسة: heritage/flood/outside_masterplan (تصعيد تلقائي).
- اليوم 3:
  - قياس SLA: زمن كل خطوة.
  - تقارير الفرع للمكتب (POST /api/survey/reports).
  - FieldApp (نقاط/أوفلاين إن لزم) + Dashboard الفرع/المكتب.

معايير النجاح
- جميع Endpoints تُعيد JSON/200 ولم يحدث 500.
- PDF فعلي يُفتح ويعرض QR وصور/حدود/اشتراطات ومصدر الهندسة.
- التصعيد يمنع التكرار ويرتد بقرار مكتب صحيح.
- SLA الرoutines ≤ 4 أيام، المصعَّد ≤ 6 أيام (محاكاة زمنياً في UAT).

أوامر Smoke النهائية
- PDF: curl -I "http://localhost:5000/api/survey/decisions/QM-TEST/download?requestId=SR-TEST-001"
- Shapefile: upload → validation → commit (كما في رسالتي السابقة).
- Escalation + office decision (كما في رسالتي السابقة).
