# 🤖 تطوير اضافي المرحلة الثاليلة : أتمتة الهيكل التنظيمي والمهام والاختصاصات

> **"نظام يقرأ الهيكل التنظيمي → يفهم المهام → يوزع المسؤوليات → يراقب الأداء — تلقائياً."**

بعد دراسة متأنية للهيكل التنظيمي المقترح لوزارة النقل والأشغال العامة، أقدم لكم تصميم متكامل لأتمتة الهيكل التنظيمي والمهام والاختصاصات.

---

## 🎯 الأهداف الاستراتيجية

1. **تحويل الهيكل التنظيمي إلى كود رقمي**: كل إدارة → وحدة تنفيذية ذكية.
2. **ربط المهام بالمسؤوليات تلقائياً**: عند إنشاء مهمة — النظام يحدد الجهة المسؤولة.
3. **واجهات No-Code لإدارة الهيكل التنظيمي**: للمدراء — لا يحتاجون مبرمجاً.
4. **محاكاة السيناريوهات**: اختبار توزيع المهام قبل التنفيذ.
5. **التعلم من التاريخ**: كل قرار بشري يُستخدم لتدريب النظام.
6. **إنشاء دليل مهام ذكي**: مرتبط مباشرة بالهيكل التنظيمي والاختصاصات.

### 1. محرك الأتمتة التنظيمية (OrganizationalAutomationEngine.ts)

```ts
// server/services/OrganizationalAutomationEngine.ts
export class OrganizationalAutomationEngine {
  async assignTask(taskData: TaskData): Promise<TaskAssignmentResult> {
    // 1. جلب جميع الإدارات والوحدات التنفيذية
    const departments = await db.department.findMany({
      include: { responsibilities: true, units: true }
    });
    
    // 2. تحديد الإدارة الأكثر ملاءمة بناءً على نوع المهمة
    const bestMatch = this.findBestDepartmentForTask(departments, taskData);
    
    if (!bestMatch) {
      return {
        assigned: false,
        message: "لم يتم العثور على إدارة مناسبة لهذه المهمة",
        suggestedActions: ["مراجعة تعريف المهمة", "إضافة إدارة جديدة للهيكل التنظيمي"]
      };
    }

    // 3. تحديد الوحدة التنفيذية الأنسب داخل الإدارة
    const bestUnit = this.findBestUnitForTask(bestMatch.units, taskData);
    
    // 4. حساب درجة الملاءمة
    const confidence = this.calculateConfidence(bestMatch, taskData);
    
    return {
      assigned: true,
      departmentId: bestMatch.id,
      departmentName: bestMatch.name,
      unitId: bestUnit?.id,
      unitName: bestUnit?.name,
      confidence,
      message: `تم تعيين المهمة إلى ${bestMatch.name}${bestUnit ? ` - ${bestUnit.name}` : ''}`,
      nextSteps: this.getNextSteps(bestMatch, bestUnit, taskData),
      recommendedActions: []
    };
  }

  private findBestDepartmentForTask(departments: Department[], taskData: TaskData): Department | null {
    // البحث عن أفضل تطابق بناءً على نوع المهمة والمسؤوليات
    let bestScore = 0;
    let bestDepartment: Department | null = null;

    for (const dept of departments) {
      const score = this.calculateDepartmentMatchScore(dept, taskData);
      
      if (score > bestScore) {
        bestScore = score;
        bestDepartment = dept;
      }
    }

    return bestDepartment;
  }

  private calculateDepartmentMatchScore(dept: Department, taskData: TaskData): number {
    let score = 0;
    
    // التحقق من التطابق مع المسؤوليات الرئيسية
    for (const responsibility of dept.responsibilities) {
      if (this.matchesResponsibility(responsibility, taskData)) {
        score += 1;
      }
    }
    
    // التحقق من التطابق مع الكلمات المفتاحية
    if (dept.keywords && taskData.keywords) {
      const keywordMatches = taskData.keywords.filter(k => 
        dept.keywords.includes(k)
      ).length;
      score += keywordMatches * 0.5;
    }
    
    return score;
  }

  private matchesResponsibility(responsibility: Responsibility, taskData: TaskData): boolean {
    // التحقق من التطابق مع أنواع المهام
    if (responsibility.taskTypes && taskData.type) {
      if (responsibility.taskTypes.includes(taskData.type)) {
        return true;
      }
    }
    
    // التحقق من التطابق مع القطاعات
    if (responsibility.sectors && taskData.sector) {
      if (responsibility.sectors.includes(taskData.sector)) {
        return true;
      }
    }
    
    // التحقق من التطابق مع الكلمات المفتاحية
    if (responsibility.keywords && taskData.keywords) {
      for (const keyword of taskData.keywords) {
        if (responsibility.keywords.includes(keyword)) {
          return true;
        }
      }
    }
    
    return false;
  }

  private findBestUnitForTask(units: Unit[], taskData: TaskData): Unit | null {
    if (!units || units.length === 0) {
      return null;
    }
    
    let bestScore = 0;
    let bestUnit: Unit | null = null;

    for (const unit of units) {
      const score = this.calculateUnitMatchScore(unit, taskData);
      
      if (score > bestScore) {
        bestScore = score;
        bestUnit = unit;
      }
    }

    return bestUnit;
  }

  private calculateUnitMatchScore(unit: Unit, taskData: TaskData): number {
    let score = 0;
    
    // التحقق من التطابق مع أنواع المهام
    if (unit.taskTypes && taskData.type) {
      if (unit.taskTypes.includes(taskData.type)) {
        score += 1;
      }
    }
    
    // التحقق من التطابق مع الكلمات المفتاحية
    if (unit.keywords && taskData.keywords) {
      const keywordMatches = taskData.keywords.filter(k => 
        unit.keywords.includes(k)
      ).length;
      score += keywordMatches * 0.3;
    }
    
    // التحقق من التطابق مع الموقع الجغرافي
    if (unit.location && taskData.location) {
      if (unit.location === taskData.location) {
        score += 0.5;
      }
    }
    
    return score;
  }

  private calculateConfidence(department: Department, taskData: TaskData): number {
    const maxPossibleScore = department.responsibilities.length + 
      (department.keywords?.length || 0) * 0.5;
    
    const actualScore = this.calculateDepartmentMatchScore(department, taskData);
    
    return Math.min(actualScore / maxPossibleScore, 1);
  }

  private getNextSteps(department: Department, unit: Unit | null, taskData: TaskData): NextStep[] {
    const steps: NextStep[] = [];
    
    // الخطوة 1: إرسال المهمة إلى الإدارة/الوحدة
    steps.push({
      step: 1,
      title: "إرسال المهمة",
      description: `تم إرسال المهمة إلى ${department.name}${unit ? ` - ${unit.name}` : ''}`,
      estimatedTime: "فوري"
    });

    // الخطوة 2: تعيين مسؤول مباشر
    steps.push({
      step: 2,
      title: "تعيين مسؤول مباشر",
      description: "سيتم تعيين مسؤول مباشر للمتابعة خلال 24 ساعة",
      estimatedTime: "1 يوم عمل"
    });

    // الخطوة 3: بدء التنفيذ
    steps.push({
      step: 3,
      title: "بدء التنفيذ",
      description: "ستبدأ عملية التنفيذ وفقاً للخطة المعتمدة",
      estimatedTime: taskData.estimatedTime || "يتم تحديده لاحقاً"
    });

    return steps;
  }
}

interface TaskData {
  id?: string;
  title: string;
  description: string;
  type: string; // 'construction', 'maintenance', 'planning', 'inspection', etc.
  sector: string; // 'transport', 'public_works', 'urban_planning', etc.
  location: string;
  keywords: string[];
  priority: 'low' | 'medium' | 'high';
  estimatedTime?: string;
  budget?: number;
}

interface TaskAssignmentResult {
  assigned: boolean;
  departmentId?: string;
  departmentName?: string;
  unitId?: string;
  unitName?: string;
  confidence: number;
  message: string;
  nextSteps: NextStep[];
  recommendedActions: string[];
}

interface NextStep {
  step: number;
  title: string;
  description: string;
  estimatedTime: string;
}

interface Department {
  id: string;
  name: string;
  code: string;
  parentDepartmentId?: string;
  level: number;
  responsibilities: Responsibility[];
  units: Unit[];
  keywords?: string[];
}

interface Responsibility {
  id: string;
  title: string;
  description: string;
  taskTypes?: string[];
  sectors?: string[];
  keywords?: string[];
}

interface Unit {
  id: string;
  name: string;
  code: string;
  departmentId: string;
  location: string;
  taskTypes?: string[];
  keywords?: string[];
}
```

```tsx
function renderDepartmentTree(departments: Department[], parentId: string | null = null, level = 0): JSX.Element {
  const children = departments.filter(dept => dept.parentDepartmentId === parentId);
  
  if (children.length === 0) {
    return <></>;
  }

  return (
    <>
      {children.map(dept => (
        <div key={dept.id} className="mb-2">
          <div 
            className="flex items-center cursor-pointer hover:bg-gray-100 p-2 rounded-md"
            style={{ paddingLeft: `${level * 20}px` }}
            onClick={() => handleDepartmentClick(dept)}
          >
            <span className="font-medium">{dept.name}</span>
            <span className="text-sm text-gray-500 mr-2">({dept.code})</span>
            {dept.units && dept.units.length > 0 && (
              <span className="text-xs bg-blue-100 text-blue-800 px-2 py-1 rounded-full mr-2">
                {dept.units.length} وحدات
              </span>
            )}
            {dept.responsibilities && dept.responsibilities.length > 0 && (
              <span className="text-xs bg-green-100 text-green-800 px-2 py-1 rounded-full">
                {dept.responsibilities.length} مسؤوليات
              </span>
            )}
          </div>
          {renderDepartmentTree(departments, dept.id, level + 1)}
        </div>
      ))}
    </>
  );
}
```

```tsx
          onClick={assignTask}
          disabled={isLoading}
          className="bg-blue-500 text-white px-6 py-3 rounded-md font-semibold w-full"
        >
          {isLoading ? 'جاري التعيين...' : 'تعيين المهمة تلقائياً'}
        </button>
      </div>

      {/* Assignment Result */}
      {assignmentResult && (
        <div className={`p-6 rounded-md ${
          assignmentResult.assigned ? 'bg-green-50 border border-green-200' : 'bg-red-50 border border-red-200'
        }`}>
          <h3 className="text-lg font-semibold mb-4">
            {assignmentResult.assigned ? '🎉 تم تعيين المهمة بنجاح' : '❌ لم يتم تعيين المهمة'}
          </h3>
          
          <p className="mb-4">{assignmentResult.message}</p>
          
          {assignmentResult.assigned && (
            <>
              <div className="mb-4">
                <h4 className="font-semibold mb-2">درجة الثقة:</h4>
                <div className="w-full bg-gray-200 rounded-full h-2.5">
                  <div 
                    className="bg-green-600 h-2.5 rounded-full" 
                    style={{ width: `${assignmentResult.confidence * 100}%` }}
                  ></div>
                </div>
                <p className="text-sm text-gray-600 mt-1">{(assignmentResult.confidence * 100).toFixed(0)}%</p>
              </div>
              
              <div className="mb-4">
                <h4 className="font-semibold mb-2">الخطوات التالية:</h4>
                <ol className="list-decimal list-inside">
                  {assignmentResult.nextSteps.map((step, i) => (
                    <li key={i} className="mb-2">
                      <strong>{step.step}. {step.title}</strong>
                      <p>{step.description}</p>
                      <p className="text-sm text-gray-600">الوقت المقدر: {step.estimatedTime}</p>
                    </li>
                  ))}
                </ol>
              </div>
              
              <button className="bg-green-500 text-white px-6 py-3 rounded-md font-semibold">
                عرض تفاصيل الإدارة/الوحدة
              </button>
            </>
          )}

          {!assignmentResult.assigned && (
            <>
              <div className="mb-4">
                <h4 className="font-semibold mb-2">الإجراءات المقترحة:</h4>
                <ul className="list-disc list-inside">
                  {assignmentResult.recommendedActions.map((action, i) => (
                    <li key={i}>{action}</li>
                  ))}
                </ul>
              </div>
              
              <button className="bg-blue-500 text-white px-6 py-3 rounded-md font-semibold">
                تعديل وصف المهمة
              </button>
            </>
          )}
        </div>
      )}
    </div>
  );
}

// Keyword Input Component
function KeywordInput({ keywords, onChange }: { keywords: string[], onChange: (keywords: string[]) => void }) {
  const [inputValue, setInputValue] = useState('');

  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Enter' && inputValue.trim()) {
      e.preventDefault();
      onChange([...keywords, inputValue.trim()]);
      setInputValue('');
    }
  };

  const removeKeyword = (index: number) => {
    const newKeywords = [...keywords];
    newKeywords.splice(index, 1);
    onChange(newKeywords);
  };

  return (
    <div>
      <div className="flex flex-wrap gap-2 mb-2">
        {keywords.map((keyword, index) => (
          <span
            key={index}
            className="inline-flex items-center px-3 py-1 rounded-full text-sm font-medium bg-blue-100 text-blue-800"
          >
            {keyword}
            <button
              type="button"
              className="ml-1 inline-flex items-center justify-center w-4 h-4 text-blue-500 hover:text-blue-700"
              onClick={() => removeKeyword(index)}
            >
              ×
            </button>
          </span>
        ))}
      </div>
      <input
        type="text"
        value={inputValue}
        onChange={(e) => setInputValue(e.target.value)}
        onKeyDown={handleKeyDown}
        className="w-full p-2 border border-gray-300 rounded-md"
        placeholder="أدخل كلمة مفتاحية ثم اضغط Enter"
      />
    </div>
  );
}
```

## 🚀 أفكار إضافية لتطوير المرحلة

### 1. "نظام التوصيات الذكية"
- بناءً على أداء الوحدات التنفيذية السابقة → النظام يقترح أفضل وحدة لكل مهمة.

### 2. "لوحة مراقبة الأداء التنظيمي"
- تتبع أداء الإدارات والوحدات التنفيذية → تحديد نقاط القوة والضعف.

### 3. "التنبيهات الذكية للتأخير"
- إذا تأخرت وحدة تنفيذية عن مواعيد التسليم → النظام ينبه الإدارة العليا.

### 4. "تحليل التكاليف التنبؤي"
- تقدير التكاليف بناءً على نوع المهمة والموقع والوحدة التنفيذية.

### 5. "التقارير التلقائية"
- إنشاء تقارير دورية عن توزيع المهام والأداء دون تدخل بشري.

### 6. "نظام التقييم التلقائي"
- تقييم أداء الموظفين والوحدات التنفيذية بناءً على نتائج المهام.
