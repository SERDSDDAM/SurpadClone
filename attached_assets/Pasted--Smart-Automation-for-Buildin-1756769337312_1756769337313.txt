### 1. محرك الأتمتة القانونية (LegalAutomationEngine.ts)
```ts
// server/services/LegalAutomationEngine.ts
export class LegalAutomationEngine {
  async evaluateServiceRequest(serviceId: string, requestData: any): Promise<AutomationResult> {
    // 1. جلب الخدمة
    const service = await db.service.findUnique({ where: { id: serviceId } });
    
    // 2. جلب الاشتراطات المرتبطة بالخدمة
    const requirements = await db.serviceRequirement.findMany({
      where: { serviceId: service.id },
      include: { requirement: true }
    });

    // 3. تقييم كل اشتراط
    const evaluations = [];
    for (let sr of requirements) {
      let passes = true;
      let message = "مستوفى";

      // تقييم الاشتراط المشروط
      if (sr.is_conditional && sr.condition_details) {
        const conditionMet = this.evaluateCondition(sr.condition_details, requestData);
        if (!conditionMet) {
          passes = false;
          message = `غير مستوفى: ${sr.requirement.requirement_text} - الشرط غير متحقق`;
        }
      }

      // تقييم الاشتراط الأساسي
      if (passes) {
        passes = this.evaluateRequirement(sr.requirement, requestData);
        if (!passes) {
          message = `غير مستوفى: ${sr.requirement.requirement_text}`;
        }
      }

      evaluations.push({
        requirement_id: sr.requirement.id,
        passed: passes,
        message
      });
    }

    // 4. اتخاذ القرار
    const allPassed = evaluations.every(e => e.passed);
    return {
      approved: allPassed,
      confidence: allPassed ? 1.0 : 0.2,
      evaluations,
      recommendedActions: allPassed ? [] : ["تعديل الطلب", "تقديم مستندات إضافية"]
    };
  }

  private evaluateCondition(condition: string, data: any): boolean {
    // تحليل الشرط مثل: "في حال زيادة مساحة البناء عن 200 متر مربع"
    if (condition.includes("زيادة مساحة البناء عن")) {
      const areaThreshold = parseInt(condition.match(/\d+/)?.[0] || "0");
      return data.area_sqm > areaThreshold;
    }
    return true; // افتراضي
  }

  private evaluateRequirement(req: Requirement, data: any): boolean {
    switch (req.value_type) {
      case 'number':
        if (req.min_value && data[req.field] < req.min_value) return false;
        if (req.max_value && data[req.field] > req.max_value) return false;
        break;
      case 'boolean':
        if (req.is_mandatory && !data[req.field]) return false;
        break;
      case 'string':
        if (req.is_mandatory && !data[req.field]) return false;
        break;
    }
    return true;
  }
}
```

### 2. واجهة "مساعد إنشاء القوانين" (No-Code Rule Builder)

#### التصميم:
```tsx
// client/src/components/RuleBuilder.tsx
function RuleBuilder() {
  const [rules, setRules] = useState([]);

  const addRule = () => {
    setRules([...rules, { field: '', operator: '', value: '' }]);
  };

  const updateRule = (index, field, value) => {
    const newRules = [...rules];
    newRules[index][field] = value;
    setRules(newRules);
  };

  const testRule = async () => {
    const testData = {
      area_sqm: 150,
      height_m: 10,
      location: 'SA-HAMDAN'
    };
    
    const result = await fetch('/api/automation/evaluate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ rules, testData })
    }).then(r => r.json());

    alert(`النتيجة: ${result.approved ? '✅ موافقة' : '❌ رفض'}`);
  };

  return (
    <div className="rule-builder p-6 bg-white rounded-lg shadow">
      <h3 className="text-xl font-bold mb-4">مساعد إنشاء القوانين</h3>
      
      {rules.map((rule, index) => (
        <div key={index} className="flex gap-2 mb-4">
          <select 
            value={rule.field} 
            onChange={(e) => updateRule(index, 'field', e.target.value)}
            className="border p-2 rounded"
          >
            <option value="">اختر الحقل</option>
            <option value="area_sqm">مساحة البناء</option>
            <option value="height_m">الارتفاع</option>
            <option value="location">الموقع</option>
          </select>
          
          <select 
            value={rule.operator} 
            onChange={(e) => updateRule(index, 'operator', e.target.value)}
            className="border p-2 rounded"
          >
            <option value="<">&lt;</option>
            <option value=">">&gt;</option>
            <option value="==">=</option>
          </select>
          
          <input 
            type="text" 
            value={rule.value} 
            onChange={(e) => updateRule(index, 'value', e.target.value)}
            placeholder="القيمة"
            className="border p-2 rounded flex-1"
          />
        </div>
      ))}

      <div className="flex gap-4">
        <button 
          onClick={addRule}
          className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
        >
          + إضافة شرط
        </button>
        
        <button 
          onClick={testRule}
          className="bg-green-500 text-white px-4 py-2 rounded hover:bg-green-600"
        >
          اختبار القاعدة
        </button>
      </div>
    </div>
  );
}
```

### 3. واجهة "محاكي السيناريوهات" (Scenario Simulator)

#### التصميم:
```tsx
// client/src/components/ScenarioSimulator.tsx
function ScenarioSimulator() {
  const [service, setService] = useState('');
  const [data, setData] = useState({});
  const [result, setResult] = useState(null);

  const runSimulation = async () => {
    const response = await fetch(`/api/automation/simulate?serviceId=${service}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    setResult(await response.json());
  };

  return (
    <div className="scenario-simulator p-6 bg-white rounded-lg shadow">
      <h3 className="text-xl font-bold mb-4">محاكي السيناريوهات</h3>
      
      <select 
        value={service} 
        onChange={(e) => setService(e.target.value)}
        className="border p-2 rounded w-full mb-4"
      >
        <option value="">اختر الخدمة</option>
        <option value="building-permit">ترخيص بناء جديد</option>
        <option value="land-subdivision">اعتماد مخطط تقسيم أراضي</option>
        <option value="land-use-change">تعديل استخدام أرض</option>
      </select>

      <div className="grid grid-cols-2 gap-4 mb-4">
        <input 
          name="area_sqm" 
          placeholder="مساحة البناء (م²)" 
          onChange={(e) => setData({...data, area_sqm: parseFloat(e.target.value)})}
          className="border p-2 rounded"
        />
        <input 
          name="height_m" 
          placeholder="الارتفاع (م)" 
          onChange={(e) => setData({...data, height_m: parseFloat(e.target.value)})}
          className="border p-2 rounded"
        />
        <select 
          name="location" 
          onChange={(e) => setData({...data, location: e.target.value})}
          className="border p-2 rounded col-span-2"
        >
          <option value="">اختر الموقع</option>
          <option value="SA-HAMDAN">همدان</option>
          <option value="AD-CRAITER">كريتر</option>
        </select>
      </div>

      <button 
        onClick={runSimulation}
        className="bg-purple-500 text-white px-4 py-2 rounded hover:bg-purple-600 w-full"
      >
        تشغيل المحاكاة
      </button>

      {result && (
        <div className="mt-4 p-4 bg-gray-50 rounded">
          <h4 className="font-bold">نتيجة المحاكاة:</h4>
          <p className={result.approved ? 'text-green-600' : 'text-red-600'}>
            {result.approved ? '✅ الموافقة' : '❌ الرفض'}
          </p>
          <ul className="mt-2">
            {result.evaluations.map((e, i) => (
              <li key={i} className={e.passed ? 'text-green-500' : 'text-red-500'}>
                • {e.message}
              </li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}
```

### 4. الجداول الجديدة في قاعدة البيانات

#### automated_decisions
```sql
CREATE TABLE automated_decisions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  service_request_id VARCHAR(255) NOT NULL,
  service_id VARCHAR(255) NOT NULL,
  decision_result VARCHAR(50) NOT NULL, -- 'approved', 'rejected'
  confidence_score DECIMAL(3,2) NOT NULL,
  evaluations JSONB, -- [{"requirement_id": "...", "passed": true, "message": "..."}]
  input_data JSONB, -- بيانات الطلب الأصلية
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

#### automation_rules
```sql
CREATE TABLE automation_rules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  rule_name VARCHAR(255) NOT NULL UNIQUE,
  service_id VARCHAR(255) NOT NULL, -- أي خدمة تنطبق عليها
  conditions JSONB NOT NULL, -- {"field": "area_sqm", "operator": "<", "value": 200}
  action JSONB NOT NULL, -- {"type": "approve", "message": "تمت الموافقة تلقائياً"}
  priority INTEGER DEFAULT 5,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

#### ai_learning_data
```sql
CREATE TABLE ai_learning_data (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  model_type VARCHAR(100) NOT NULL, -- 'approval_model', 'risk_model'
  input_data JSONB NOT NULL, -- بيانات الطلب الأصلية
  expected_output JSONB, -- ما كان يجب أن يكون القرار
  actual_output JSONB, -- ما قرره النظام
  accuracy_score DECIMAL(3,2),
  feedback_provided BOOLEAN DEFAULT FALSE,
  learning_session VARCHAR(255), -- 'session-2025-06'
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

## 💡 أفكار إضافية لتطوير المرحلة

### 1. "الوضع الاستباقي" في التطبيق المحمول
- عندما يفتح المواطن التطبيق في موقع معين → النظام يقترح عليه الخدمات المناسبة لهذا الموقع تلقائياً.

### 2. "المساعد الصوتي القانوني"
- مواطن يسأل: "هل يمكنني بناء دور ثالث هنا؟" → النظام يجيب: "نعم، بشرط أن يكون الارتفاع أقل من 12 مترًا حسب المادة 15 من قانون البناء."

### 3. "التنبيهات الوقائية"
- إذا بدأ المواطن في رفع ملفات لمشروع تجاري → النظام ينبهه: "لا تنسَ توفير موقف سيارات كما يشترط القانون."

### 4. "تحليل المخالفات التنبؤي"
- النظام يحلل طلبات المناطق ذات كثرة المخالفات → يقترح زيادة الرقابة فيها.
